\section{Contexte}

\begin{frame}{Pourquoi ce sujet ?}
  \begin{columns}[c]
    \begin{column}{0.55\textwidth}
      \footnotesize
      \begin{itemize}
        \item Un contrat intelligent est un \textbf{programme public, déterministe et immuable} : une faille déployée ne se corrige pas (Ch08).
        \item \textbf{The DAO} (2016) : une \textbf{réentrance} détourne environ 3,6 M d'ETH. La bifurcation dure qui suit fait naître Ethereum Classic \refcite{atzei2017}.
        \item Enjeu local : un diplôme de l'ESP vérifié sur une chaîne de blocs (sujet A13) reposerait lui aussi sur des contrats.
      \end{itemize}
    \end{column}
    \begin{column}{0.43\textwidth}

\begin{frame}{Question de recherche}
  \begin{bloc}{esporange}{Notre question (fiche officielle A11)}
    Les outils automatiques (Slither, Mythril, Echidna) détectent-ils les vulnérabilités réellement exploitées, et avec quel taux de faux positifs sur des contrats récents (Solidity 0.8) ?
  \end{bloc}
  \footnotesize
  \begin{itemize}
    \item \textbf{Hypothèse} : les outils sont complémentaires. Leur couverture dépend de la méthode et de la catégorie de faille. Les combiner l'améliore, sans garantir l'absence de faille.
    \item \textbf{Méthode envisagée} : au moins 20 contrats annotés (Solidity 0.8 en priorité), une matrice de confusion, puis précision, rappel et temps d'analyse, comparés à \refcite{durieux2020}.
    \item \textbf{Modèle d'attaque} : adversaire actif (Modou). Il déploie son propre contrat, appelle les fonctions publiques et exécute du code à chaque réception d'ETH, sans aucune clé privée d'autrui.
  \end{itemize}
\end{frame}

\section{Mécanisme}

\begin{frame}{Principe du mécanisme}
  \begin{bloc}{espblue}{Analyse automatique d'un contrat}
    \small \textbf{Entrée} : code Solidity ou bytecode, plus des propriétés pour Echidna. \textbf{Sortie} : des alertes (catégorie, emplacement, gravité) ou un contre-exemple. \textbf{Propriété visée} (réentrance) : l'état est mis à jour \emph{avant} tout appel externe, et la somme des soldes reste au plus égale aux ETH détenus.
  \end{bloc}
  \vspace{2pt}
  \centering
  \begin{tikzpicture}[>=Stealth, thick, font=\scriptsize,
      boite/.style={draw=espblue, fill=espblue!6, rounded corners, minimum height=0.62cm, align=center},
      outil/.style={draw=esporange, fill=esporange!10, rounded corners, minimum width=3.6cm, minimum height=0.52cm, align=center}]
    \node[boite, minimum width=2.4cm] (src) at (0,0) {contrat \texttt{.sol}\\(+ propriétés)};
    \node[outil] (sl) at (4.2,0.72) {Slither --- analyse statique};
    \node[outil] (my) at (4.2,0)    {Mythril --- exécution symbolique};
    \node[outil] (ec) at (4.2,-0.72){Echidna --- test aléatoire (\emph{fuzzing})};
    \node[boite, minimum width=2.3cm] (out) at (8.3,0) {alertes /\\contre-exemples};
    \node[boite, minimum width=2.3cm] (rev) at (11.3,0) {revue manuelle\\VP · FP · FN};
    \foreach \t in {sl,my,ec} { \draw[->] (src.east) -- (\t.west); \draw[->] (\t.east) -- (out.west); }
    \draw[->] (out) -- (rev);
  \end{tikzpicture}
\end{frame}

\begin{frame}{Exemple déroulé}
  \begin{bloc}{espgreen}{Calcul pas à pas (mesuré sur le réseau Hardhat local)}
    \small
    \begin{enumerate}
      \item \textbf{Paramètres} : Alice dépose 10 ETH, Modou mise 1 ETH, donc le coffre contient 11 ETH.
      \item \textbf{Étape 1} : \texttt{retirer()} lit \texttt{montant = 1} et envoie 1 ETH \emph{avant} \texttt{soldes[Modou] = 0}. La fonction \texttt{receive()} de Modou rappelle alors \texttt{retirer()}, et son solde enregistré vaut \emph{toujours} 1.
      \item \textbf{Étape 2} : chaque réentrée retire 1 ETH. Le coffre passe de 10 à 9, puis 8, et ainsi de suite jusqu'à 0, soit \textbf{10 réentrées}.
      \item \textbf{Résultat et vérification} : Modou détient 11 ETH (1 + 10), et le coffre est vide. Pourtant, le registre indique encore 10 ETH pour Alice, qui ne peut plus retirer.
    \end{enumerate}
  \end{bloc}
\end{frame}

\begin{frame}{Ce qui peut mal tourner}
  \begin{bloc}{espred}{Mauvais usage ou attaque connue}
    \footnotesize
    \textbf{Réentrance} : appel externe avant la mise à jour de l'état. Solidity 0.8 supprime les dépassements silencieux, \emph{pas} la réentrance. \textbf{Impact} : perte ou gel de fonds, sans retour possible.
    \vspace{3pt}

    \begin{columns}[T]
      \begin{column}{0.48\textwidth}
        \textbf{Taxonomie d'Atzei et al.} \refcite{atzei2017}
        \begin{itemize}
          \item dépassement arithmétique (vérifié depuis 0.8, sauf \texttt{unchecked}) ;
          \item contrôle d'accès (Parity, 2017 : 513\,774 ETH gelés) ;
          \item dépendance à l'horodatage ;
          \item exécution anticipée (\emph{front-running}).
        \end{itemize}
      \end{column}
      \begin{column}{0.48\textwidth}
        \textbf{Du côté des outils}
        \begin{itemize}
          \item fausse alerte : motif signalé mais non exploitable ;
          \item faille manquée : la logique métier leur échappe ;
          \item risques hors du code (oracles, ponts, clés), invisibles à l'analyse d'un fichier.
        \end{itemize}
      \end{column}
    \end{columns}
  \end{bloc}
\end{frame}

\section{État de l'art}

\begin{frame}{Travaux existants}
  \scriptsize
  \renewcommand{\arraystretch}{1.15}
  \begin{tabularx}{\textwidth}{@{}>{\raggedright\arraybackslash}p{2.9cm}>{\raggedright\arraybackslash}X>{\raggedright\arraybackslash}p{4.6cm}@{}}
    \toprule
    Référence & Apport & Limite \\
    \midrule
    Atzei et al., 2017 \refcite{atzei2017} & Taxonomie des attaques sur Ethereum (Solidity, EVM, chaîne de blocs) & Antérieure à Solidity 0.8 ; n'évalue aucun outil \\
    Tsankov et al., 2018 \refcite{tsankov2018} & Securify : analyse statique par motifs de conformité et de violation & Limité aux propriétés encodées dans ses motifs \\
    Feist et al., 2019 \refcite{feist2019} & Slither : analyse statique rapide sur une représentation intermédiaire (SlithIR) & Signale des motifs sans prouver leur exploitabilité \\
    Durieux et al., 2020 \refcite{durieux2020} & 9 outils, 69 contrats annotés : seulement 42\,\% des failles détectées par l'ensemble des outils ; 97\,\% d'environ 47\,500 contrats réels signalés & Contrats antérieurs à Solidity 0.8, d'où un biais face aux contrats récents \\
    Grieco et al., 2020 \refcite{grieco2020} & Echidna : test aléatoire guidé par des propriétés & Ne trouve que ce que décrivent les propriétés \\
    \bottomrule
  \end{tabularx}

  \vspace{4pt}
  \centering 5 sources primaires, dont les 3 lectures obligatoires (Atzei, Slither, Durieux).
\end{frame}

\begin{frame}{Positionnement de notre projet}
  \begin{itemize}
    \item \textbf{Ce qui n'a pas été étudié} : l'étude de référence \refcite{durieux2020} porte sur des contrats antérieurs à Solidity 0.8. On ignore si ses taux de détection et de faux positifs tiennent pour des contrats récents, où l'arithmétique est vérifiée par défaut.
    \item \textbf{Notre contribution} : mesurer, sur au moins 20 contrats annotés en priorité Solidity 0.8, ce que Slither, Mythril et Echidna détectent, ratent ou signalent à tort, par catégorie de faille. La vérité terrain est relue par deux membres, et le dépôt est reproductible (scripts, versions et sorties brutes).
  \end{itemize}
\end{frame}

function retirer() external {
  uint montant = soldes[msg.sender];

\section{Conclusion}

\begin{frame}{À retenir}
  \begin{bloc}{esporange}{À retenir}
    \begin{itemize}
      \item La réentrance vient de l'\textbf{ordre des instructions} : il faut mettre à jour l'état \emph{avant} tout appel externe. Solidity 0.8 ne la supprime pas.
      \item Chaque famille d'outils a un angle mort. Une alerte ne prouve pas que la faille est exploitable, et \textbf{une absence d'alerte ne prouve pas la sécurité}.
      \item Sur notre coffre, Slither a bien signalé la réentrance réelle. La réponse à la question officielle viendra du corpus d'au moins 20 contrats annotés (résultats à venir).
    \end{itemize}
  \end{bloc}
\end{frame}
